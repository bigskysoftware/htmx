import { getFolder } from '../../lib/content';
import { PATTERN_CATEGORIES, groupFilesByCategory } from '../../lib/content-groups';
import JumpTo from '../../components/JumpTo.astro';
import IndexTable from '../../components/IndexTable.astro';

export const folder = await getFolder('patterns');
export const groupedPatterns = groupFilesByCategory(folder.files, PATTERN_CATEGORIES).map(group => ({
    slug: group.slug,
    label: group.category,
    items: group.files.map(f => ({
        title: f.frontmatter.title,
        url: f.frontmatter.soon ? undefined : f.url,
        description: f.frontmatter.description,
        icon: f.frontmatter.icon || '',
        badge: f.frontmatter.soon ? 'Soon' : undefined,
    })),
}));

export const getPageHeadings = () => groupedPatterns.map(group => ({
    depth: 2,
    slug: group.slug,
    text: group.label,
}));

<label data-active-search class="not-prose text-field" for="patterns-search">
    <span class="sr-only">Search patterns</span>
    <i class="icon-[mdi--magnify]" aria-hidden="true"></i>
    <input
        id="patterns-search"
        type="text"
        placeholder="Search patterns..."
        autocomplete="off"
        autocapitalize="off"
        autocorrect="off"
        spellcheck="false"
        enterkeyhint="search"
        aria-keyshortcuts="/"
        _="install ActiveSearch"
    />
    <span aria-hidden="true">Focus <kbd>/</kbd></span>
</label>

<JumpTo headings={getPageHeadings()} />

{groupedPatterns.map(group => (
<section role="group">

<h2 id={group.slug}>
    <span>
        <a href={`#${group.slug}`} class="no-underline interact:underline">
            {group.label}
        </a>
    </span>
</h2>

<IndexTable>
    {group.items.map(item => (
        <tr class={item.badge ? 'opacity-50' : undefined}>
            <th scope="row">
                <span class="flex items-center gap-5">
                    {item.icon ? <i class={`${item.icon} icon size-5 shrink-0`} aria-hidden="true"></i> : <span class="size-5 shrink-0"></span>}
                    <span>
                        {item.url ? <a href={item.url}>{item.title}</a> : item.title}
                        {item.badge && <span class="sm:hidden text-[9px] ml-2 px-1.5 py-0.5 rounded bg-yellow-300/5 text-yellow-700 dark:text-yellow-300 border border-yellow-600/50 dark:border-yellow-300 font-medium uppercase tracking-wide">{item.badge}</span>}
                    </span>
                </span>
            </th>
            <td class="text-neutral-650 dark:text-neutral-350">
                {item.description || ''}
                {item.badge && <span class="hidden sm:inline text-[9px] ml-2 px-1.5 py-0.5 rounded bg-yellow-300/5 text-yellow-700 dark:text-yellow-300 border border-yellow-600/50 dark:border-yellow-300 font-medium uppercase tracking-wide">{item.badge}</span>}
            </td>
        </tr>
    ))}
</IndexTable>

</section>
))}
