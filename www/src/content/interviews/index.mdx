import { getFolder } from '../../lib/content';
import IndexTable from '../../components/IndexTable.astro';

export const interviewFolder = await getFolder('interviews');
export const interviews = interviewFolder.files.map(e => ({
    title: e.frontmatter.title,
    url: e.url,
    author: e.frontmatter.authors?.join(', '),
    date: e.frontmatter.created ? new Date(e.frontmatter.created).toLocaleDateString('en-US', {month: 'short', day: 'numeric', year: 'numeric'}) : undefined,
}));

<IndexTable>
    {interviews.map(item => (
        <tr>
            <th scope="row"><a href={item.url}>{item.title}</a></th>
            <td>
                <span>
                    <span>{item.author || ''}</span>
                    <time>{item.date || ''}</time>
                </span>
            </td>
        </tr>
    ))}
</IndexTable>
